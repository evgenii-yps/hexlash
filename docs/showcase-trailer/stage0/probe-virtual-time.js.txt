(() => {
  let now = 1_000_000; // virtual ms
  const DT = 16.6667;
  const realRAF = window.requestAnimationFrame.bind(window);
  let rafId = 0; let rafQ = [];
  const timers = new Map(); let tid = 1;
  window.requestAnimationFrame = (cb) => { rafQ.push([++rafId, cb]); return rafId; };
  window.cancelAnimationFrame = (id) => { rafQ = rafQ.filter(x => x[0] !== id); };
  performance.now = () => now - 1_000_000 + 0; // origin 0
  Date.now = () => 1_700_000_000_000 + (now - 1_000_000);
  const RD = Date;
  window.setTimeout = (fn, ms=0, ...a) => { const id = tid++; timers.set(id, { at: now + Math.max(0, ms), fn, a, iv: null }); return id; };
  window.setInterval = (fn, ms=0, ...a) => { const id = tid++; timers.set(id, { at: now + Math.max(1, ms), fn, a, iv: Math.max(1, ms) }); return id; };
  window.clearTimeout = window.clearInterval = (id) => { timers.delete(id); };
  // Math.random seeded
  let s = 12345; Math.random = () => { s |= 0; s = s + 0x6D2B79F5 | 0; let t = Math.imul(s ^ s >>> 15, 1 | s); t = t + Math.imul(t ^ t >>> 7, 61 | t) ^ t; return ((t ^ t >>> 14) >>> 0) / 4294967296; };
  window.__vt = {
    frame: 0,
    step(n = 1) {
      for (let i = 0; i < n; i++) {
        now += DT; this.frame++;
        // fire due timers
        for (const [id, t] of [...timers]) { if (t.at <= now) { if (t.iv) t.at += t.iv; else timers.delete(id); try { t.fn(...t.a); } catch (e) { console.error('timer', e); } } }
        const q = rafQ; rafQ = [];
        for (const [, cb] of q) { try { cb(now - 1_000_000); } catch (e) { console.error('raf', e); } }
      }
    },
    get now() { return now - 1_000_000; },
  };
})();
